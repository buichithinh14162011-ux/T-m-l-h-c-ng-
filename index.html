        function initTailwind() {

            tailwind.config = { content: [], theme: { extend: {} } };

        function addMessage(text, isUser) {

            const container = document.getElementById('messages');

            const bubble = document.createElement('div');

            bubble.className = `flex ${isUser ? 'justify-end' : 'justify-start'} message-bubble`;

            

            if (isUser) {

                bubble.innerHTML = `<div class="bg-teal-500 text-white px-5 py-3 rounded-3xl rounded-tr-none">${text}</div>`;

            } else {

                bubble.innerHTML = `

                    <div class="flex gap-3">

                        <div class="w-8 h-8 bg-teal-100 rounded-2xl flex-shrink-0 flex items-center justify-center text-xl mt-1">🌱</div>

                        <div class="bg-white shadow-sm px-5 py-3 rounded-3xl rounded-tl-none text-gray-800">${text}</div>

                    </div>`;

            }

            container.appendChild(bubble);

            container.scrollTop = container.scrollHeight;

        }


        function getAIResponse(userMessage) {

            const msg = userMessage.toLowerCase().trim();

            

            if (msg.includes("áp lực") || msg.includes("thi") || msg.includes("học")) {

                return "Mình hiểu bạn đang cảm thấy rất áp lực với việc học phải không? Áp lực học tập là chuyện nhiều bạn gặp phải. Bạn đang áp lực về điểm số hay kỳ thi sắp tới? Mình ở đây để lắng nghe. Nhớ là nghỉ ngơi hợp lý và chia sẻ với gia đình hoặc thầy cô cũng rất quan trọng nhé ❤️";

            } 

            else if (msg.includes("bạn") || msg.includes("bạn bè") || msg.includes("cãi") || msg.includes("mâu thuẫn")) {

                return "Có vẻ bạn đang gặp vấn đề với bạn bè. Mâu thuẫn là điều bình thường. Bạn có thể kể mình nghe chi tiết hơn được không? Đôi khi việc nói chuyện thẳng thắn sẽ giúp ích rất nhiều. Bạn không cô đơn đâu.";

            } 

            else if (msg.includes("động lực") || msg.includes("chán") || msg.includes("thất vọng")) {

                return "Thiếu động lực là điều nhiều bạn gặp phải. Bạn đã thử làm gì để tìm lại niềm vui chưa? Có lẽ bắt đầu từ những việc nhỏ bạn thích. Và nếu cần, hãy chia sẻ với gia đình để họ cùng động viên bạn.";

            } 

            else if (msg.includes("căng thẳng") || msg.includes("lo âu") || msg.includes("stress") || msg.includes("lo lắng") || msg.includes("buồn")) {

                return "Nghe bạn chia sẻ, mình cảm nhận được bạn đang khá căng thẳng và lo lắng. Đó là cảm xúc hoàn toàn bình thường. Bạn đã thử hít thở sâu hay đi dạo chưa? Nếu cảm giác này kéo dài, mình khuyên thật lòng bạn nên nói với thầy cô hoặc người thân. Bạn không cần phải gánh vác một mình.";

            } 

            else if (msg.includes("tương lai") || msg.includes("nghề") || msg.includes("đại học") || msg.includes("định hướng")) {

                return "Bạn đang băn khoăn về định hướng tương lai à? Đây là giai đoạn nhiều bạn lo lắng. Hãy liệt kê những thứ bạn thích và điểm mạnh của mình. Không cần phải quyết định ngay. Bạn có thể tham khảo thầy cô hoặc làm một số bài test hướng nghiệp. Mình tin bạn sẽ tìm ra con đường phù hợp!";

            }

            

            // Fallback

            return `Cảm ơn bạn đã tin tưởng chia sẻ với mình. Mình đang lắng nghe và thấu hiểu những gì bạn đang trải qua. Dù vấn đề có nhỏ hay lớn, việc nói ra đã là một bước rất dũng cảm rồi. Bạn có muốn kể thêm chi tiết không?<br><br>Nhớ nhé, bạn không cần phải tự mình gánh vác mọi thứ — hãy tìm đến bạn bè, gia đình, thầy cô hoặc chuyên gia khi cần. Mình luôn ở đây đồng hành cùng bạn ❤️`;

        }


        function sendMessage() {

            const input = document.getElementById('user-input');

            const text = input.value.trim();

            if (!text) return;

            

            addMessage(text, true);

            input.value = '';

            

            setTimeout(() => {

                const response = getAIResponse(text);

                addMessage(response, false);

            }, 800);

        }


        function startTopic(topic) {

            const container = document.getElementById('messages');

            container.innerHTML = '';

            

            let greeting = "Chào bạn! Mình rất vui khi bạn muốn nói về chủ đề này.";

            

            if (topic === 'ap-luc') greeting = "Áp lực học tập là một trong những vấn đề phổ biến nhất. Bạn đang cảm thấy áp lực như thế nào? Mình sẵn sàng lắng nghe và cùng bạn tìm cách vượt qua nhé.";

            else if (topic === 'ban-be') greeting = "Bạn đang gặp khó khăn với bạn bè phải không? Kể mình nghe đi, mình sẽ cố gắng thấu hiểu và đưa ra lời khuyên chân thành.";

            else if (topic === 'dong-luc') greeting = "Thiếu động lực học tập là chuyện rất bình thường. Bạn đang cảm thấy chán hay mệt mỏi vì học? Mình ở đây để cùng bạn tìm lại động lực.";

            else if (topic === 'can-thang') greeting = "Căng thẳng và lo âu là cảm xúc hoàn toàn bình thường. Bạn đang lo lắng điều gì? Mình lắng nghe đây, và mình khuyên bạn đừng ngần ngại tìm sự hỗ trợ từ người thân.";

            else if (topic === 'tuong-lai') greeting = "Định hướng tương lai đôi khi khiến ta bối rối. Bạn đang nghĩ gì về tương lai của mình? Mình ở đây để cùng bạn suy nghĩ tích cực nhé.";

            

            addMessage(greeting, false);

        }

Bạn không cần phải mạnh mẽ một mình.<br><br>Hôm nay bạn muốn chia sẻ gì nào?", false);

                }, 300);

            }

        }
